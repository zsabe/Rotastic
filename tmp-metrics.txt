12398: function getBriefingPriority(scheduleDay, doctorName, period) {
12399:   const shift = scheduleDay && scheduleDay[period];
12400:   if (!shift) return -1;
12401: 
12402:   let score = 0;
12403: 
12404:   if (doctorMatches(shift.timed1Doc, doctorName)) score += 100;
12405:   if (doctorMatches(shift.timed2Doc, doctorName)) score += 90;
12406: 
12407:   if (doctorMatches(shift.med, doctorName)) score += 80;
12408:   if (doctorMatches(shift.fed, doctorName)) score += 80;
12409:   if (doctorMatches(shift.ped, doctorName)) score += 85;
12410: 
12411:   const special = collectSpecialClinicDuties(shift, doctorName);
12412:   score += special.length * 70;
12413: 
12414:   const rooms = getRoomAssignmentsForDoctor(shift, doctorName);
12415:   score += rooms.length * 45;
12416: 
12417:   const cb = period === 'am' ? scheduleDay.cbAM : scheduleDay.cbPM;
12418:   if (cb) {
12419:     if (doctorMatches(cb.tl, doctorName)) score += 40;
12420:     if (doctorMatches(cb.tm2, doctorName) || doctorMatches(cb.tm3, doctorName)) score += 25;
12421:   }
12422: 
12423:   const lead = scheduleDay[period === 'am' ? 'shiftLeadAM' : 'shiftLeadPM'];
12424:   if (doctorMatches(lead, doctorName)) score += 50;
12425: 
12426:   return score;
12427: }

12429: function chooseSingleDailyBriefing(scheduleDay, doctorName) {
12430:   const am = buildShiftBriefing(scheduleDay, doctorName, 'am');
12431:   const pm = buildShiftBriefing(scheduleDay, doctorName, 'pm');
12432: 
12433:   if (am && !pm) return am;
12434:   if (pm && !am) return pm;
12435:   if (!am && !pm) return null;
12436: 
12437:   // Defensive rule: doctors must never work both AM and PM on the same day.
12438:   // If legacy rota data accidentally places a doctor in both, show only the
12439:   // more clinically significant assignment instead of presenting two shifts.
12440:   const amScore = getBriefingPriority(scheduleDay, doctorName, 'am');
12441:   const pmScore = getBriefingPriority(scheduleDay, doctorName, 'pm');
12442: 
12443:   if (pmScore > amScore) return pm;
12444:   return am;
12445: }
