function getDoctorShiftTiming(scheduleDay, doctorName) {
  if (!scheduleDay || !doctorName) return null;

  const candidates = ['am', 'pm'].map(function(period) {
    const briefing = buildShiftBriefing(scheduleDay, doctorName, period);
    if (!briefing) return null;

    const shift = scheduleDay[period] || {};
    const isAM = period === 'am';
    const isTimed = doctorMatches(shift.timed1Doc, doctorName) || doctorMatches(shift.timed2Doc, doctorName);

    return {
      period: period,
      label: isTimed
        ? (isAM ? '09:00–16:00' : '14:00–20:00')
        : (isAM ? '07:00–14:00' : '16:00–22:00'),
      startMinutes: isTimed ? (isAM ? 9 * 60 : 14 * 60) : (isAM ? 7 * 60 : 16 * 60),
      endMinutes: isTimed ? (isAM ? 16 * 60 : 20 * 60) : (isAM ? 14 * 60 : 22 * 60),
      briefing: briefing
    };
  }).filter(Boolean);

  if (!candidates.length) return null;
  if (candidates.length === 1) return candidates[0];

  // Defensive: if legacy data places a doctor in both periods, mirror the same
  // priority logic used by the daily briefing.
  const amScore = getBriefingPriority(scheduleDay, doctorName, 'am');
  const pmScore = getBriefingPriority(scheduleDay, doctorName, 'pm');
  return pmScore > amScore
    ? candidates.find(function(c) { return c.period === 'pm'; })
    : candidates.find(function(c) { return c.period === 'am'; });
}
