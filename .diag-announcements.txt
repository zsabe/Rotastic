function nextGreetingVibe() {
  if (!ROTASTIC_GREETING_VIBES.length) return '';

  let options = ROTASTIC_GREETING_VIBES.filter(function(msg) {
    return msg !== rotasticLastGreetingVibe;
  });

  if (!options.length) options = ROTASTIC_GREETING_VIBES.slice();

  const message = options[Math.floor(Math.random() * options.length)];
  rotasticLastGreetingVibe = message;
  return message;
}

function buildRecurringDoctorGreeting() {
  const doctor = String(activeUserDocName || '').replace(/^Dr\.?\s*/i, '').trim();
  const first = doctor.split(/\s+/)[0] || 'Doctor';

  let hour = new Date().getHours();
  try {
    hour = parseInt(
      new Intl.DateTimeFormat('en-GB', {
        timeZone:'Asia/Qatar',
        hour:'2-digit',
        hour12:false
      }).format(new Date()),
      10
    );
  } catch (e) {}

  const greeting =
    hour < 12 ? 'Good morning' :
    hour < 17 ? 'Good afternoon' :
    'Good evening';

  const wish = nextGreetingVibe();
  return greeting + ', Dr ' + first + '.\n' + wish;
}

async function fetchRegionalMedicalSource(source) {
  try {
    const items = await fetchRssJson(regionalMedicalFeedUrl(source.query));

    return items
      .map(function(item) {
        return {
          title: normalizeFeedTitle(item.title),
          link: String(item.link || item.url || source.home || ''),
          sourceName: source.name,
          domain: source.domain,
          priority: source.priority || 1
        };
      })
      .filter(function(item) {
        return item.title &&
          item.title.length > 18 &&
          isRelevantFamilyMedicineHeadline(item.title, item.sourceName);
      })
      .map(function(item) {
        item.title = cleanInterestingMedicalTickerText(item.title, item.sourceName);
        return item;
      })
      .slice(0,6);
  } catch (e) {
    return [];
  }
}

function cleanRegionalMedicalHeadline(title, sourceName) {
  let value = String(title || '')
    .replace(/\s+-\s+Google News\s*$/i, '')
    .replace(/\s+-\s+[^-]{2,45}$/i, function(match) {
      const source = String(sourceName || '').toLowerCase();
      const tail = match.toLowerCase();
      return tail.includes(source.split(' ')[0]) ? '' : match;
    })
    .replace(/\s+/g, ' ')
    .trim();
  return fitNormalTickerCopy(value, 15, 108);
}

function fitNormalTickerCopy(value, maxWords, maxChars) {
  const text = String(value || '').replace(/\s+/g, ' ').trim();
  if (!text || text.includes('…') || text.includes('...')) return '';
  const words = text.split(/\s+/).filter(Boolean);
  if (words.length > Number(maxWords || 16)) return '';
  if (text.length > Number(maxChars || 112)) return '';
  return text;
}

function getTickerCopySizeClass(message) {
  const text = String(message || '').trim();
  const words = text.split(/\s+/).filter(Boolean).length;
  if (words <= 8 && text.length <= 58) return 'ticker-copy-short';
  if (words <= 12 && text.length <= 86) return 'ticker-copy-medium';
  return 'ticker-copy-long';
}

}

  } else if ((type === 'medicalnews' && rotasticCurrentMedicalLink) || (['qnanews','worldnews','alarabiyanews','reutersnews'].includes(type) && rotasticCurrentTickerLink)) {
    const link = document.createElement('a');
    link.className = 'ticker-medical-link';
    link.href = type === 'medicalnews' ? rotasticCurrentMedicalLink : rotasticCurrentTickerLink;
    link.target = '_blank';
    link.rel = 'noopener noreferrer';
    link.textContent = message;
    link.title = 'Open source';
    card.appendChild(link);
  } else {
    card.textContent = message;
  }
  content.appendChild(card);
  if (headlineTickerTimer) clearTimeout(headlineTickerTimer);
  const requestedDuration = Number(durationMs || 0);
  const comfortableDuration = getTickerDisplayDuration(message, 7800);
  const finalDuration = Math.max(requestedDuration, comfortableDuration);
  headlineTickerTimer = setTimeout(function() { renderNextTickerSlot(); }, finalDuration);
  scheduleSickTakeoverInterrupt();
}


function renderTickerCard(type, message, durationMs, fromHistory) {
  const immediate = type === 'sick' || type === 'greeting' || type === 'welcome';
  rotasticRunTickerTransition(function() {
    renderTickerCardNow(type, message, durationMs, fromHistory);
  }, immediate);
}

function choosePrayerMessage(prayerTimes) {
  const list = getRelevantPrayerMessages(prayerTimes);
  if (!list.length) return '';
  const message = list[prayerTickerIndex % list.length];
  prayerTickerIndex = (prayerTickerIndex + 1) % list.length;
  return message;
}

let rotasticLastGreetingVibe = '';

function nextGreetingVibe() {
  if (!ROTASTIC_GREETING_VIBES.length) return '';

  let options = ROTASTIC_GREETING_VIBES.filter(function(msg) {
    return msg !== rotasticLastGreetingVibe;
  });

  if (!options.length) options = ROTASTIC_GREETING_VIBES.slice();

  const message = options[Math.floor(Math.random() * options.length)];
  rotasticLastGreetingVibe = message;
  return message;
}

function buildRecurringDoctorGreeting() {
  const doctor = String(activeUserDocName || '').replace(/^Dr\.?\s*/i, '').trim();
  const first = doctor.split(/\s+/)[0] || 'Doctor';

  let hour = new Date().getHours();
  try {
    hour = parseInt(
      new Intl.DateTimeFormat('en-GB', {
        timeZone:'Asia/Qatar',
        hour:'2-digit',
        hour12:false
      }).format(new Date()),
      10
    );
  } catch (e) {}

  const greeting =
    hour < 12 ? 'Good morning' :
    hour < 17 ? 'Good afternoon' :
    'Good evening';

  const wish = nextGreetingVibe();
  return greeting + ', Dr ' + first + '.\n' + wish;
}

async function renderNextTickerSlot() {

  if (!profileSetupIsComplete()) {
    renderPersistentFirstSetupTicker();
    return;
  }

  const launchNotification = rotasticTakeLaunchTickerNotification();
  if (launchNotification) {
    rotasticMarkNotificationShownOnce(launchNotification);
    const launchText = '🔔 ' + String(launchNotification.message || 'You have a new ROTASTIC notification.');
    renderTickerCard('notification', launchText, 15000);
    return;
  }

  // After all launch notifications have appeared once, preserve the standard
  // greeting → centre identity → normal announcement sequence.
  if (!rotasticOpeningSequenceHasRun && !rotasticOpeningTickerActive) {
    startOpeningTickerSequence();
    return;

    fetchAbuNakhlaPrayerTimes(),

    refreshExternalHeadlineFeeds()

    ? tickerFetches[0].value

  const meta = getActiveTickerMeta();

  // Live rota changes are highlighted in the same top ticker with a distinct style.
  const rotaAlert = Storage.getJSON('rotastic_rota_change_alert', null);
  if (rotaAlert && rotaAlert.message && Number(rotaAlert.expiresAt || 0) > Date.now()) {
    if (tickerSlotCounter % 3 === 1) {
      renderTickerCard('rota', rotaAlert.message, 8500);
