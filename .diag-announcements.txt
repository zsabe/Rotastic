function shuffleArray(items) {
  const arr = Array.isArray(items) ? items.slice() : [];
  for (let i = arr.length - 1; i > 0; i--) {
    let j;
    try {
      const random = new Uint32Array(1);
      crypto.getRandomValues(random);
      j = random[0] % (i + 1);
    } catch (e) {
      j = Math.floor(Math.random() * (i + 1));
    }
    const temp = arr[i];
    arr[i] = arr[j];
    arr[j] = temp;
  }
  return arr;
}

function getTrafficWindowForCurrentUser() {
  const shift = getActiveUserShiftWindowToday();
  if (!shift) return null;

  const now = getQatarClock().minutes;
  const start = shift.start;
  const end = shift.end;

  // Arrival window: 60 min before start until 20 min after start.
  if (now >= start - 60 && now <= start + 20) {
    return {
      phase:'arrival',
      shiftLabel:shift.label,
      minutesToEvent:start - now
    };
  }

  // End-of-shift window: 60 min before finish until 20 min after finish.
  if (now >= end - 60 && now <= end + 20) {
    return {
      phase:'departure',
      shiftLabel:shift.label,
      minutesToEvent:end - now
    };
  }

  return null;
}

function normalizeTrafficItem(item, phase) {
  if (!item) return null;

  if (typeof item === 'string') {
    return {text:item.trim(), phase:phase || '', source:''};
  }

  const text = String(item.text || item.summary || item.message || '').trim();
  const source = String(item.source || '').trim();
  const delayMinutes = Number(item.delayMinutes || item.delay || 0);

  if (!text) return null;

  return {
    text:text,
    phase:phase || String(item.phase || ''),
    source:source,
    delayMinutes:isNaN(delayMinutes) ? 0 : delayMinutes
  };
}
