function prayerTimeToMinutes(value) {
  if (!value) return null;
  const raw = String(value).trim();
  const ampm = /\b(AM|PM)\b/i.exec(raw);
  const parts = raw.replace(/\s*(AM|PM).*/i, '').split(':');
  if (parts.length < 2) return null;
  let h = parseInt(parts[0],10);
  const min = parseInt(parts[1],10);
  if (isNaN(h) || isNaN(min)) return null;
  if (ampm) {
    const isPM = ampm[1].toUpperCase() === 'PM';
    h = h % 12;
    if (isPM) h += 12;
  }
  return h * 60 + min;
}

function getActiveUserShiftWindowToday() {
  const today = getTodayScheduleDay();
  if (!today || !activeUserDocName) return null;

  const briefing = chooseSingleDailyBriefing(today, activeUserDocName);
  if (!briefing || !briefing.title) return null;

  const title = briefing.title;

    return {start:9*60, end:16*60, label:'AM Twilight 9–4'};
  }
  if (title.indexOf('14:00–20:00') !== -1 || title.indexOf('PM Twilight') !== -1) {
    return {start:14*60, end:20*60, label:'PM Twilight 2–8'};
  }
  if (title.indexOf('AM shift') !== -1) {

  }
  if (title.indexOf('PM shift') !== -1) {
    return {start:16*60, end:22*60, label:'PM shift'};
  }
  return null;
}

async function fetchAbuNakhlaPrayerTimes() {
  const now = getQatarClock();

  if (abuNakhlaPrayerTimes && abuNakhlaPrayerFetchDate === now.isoDate) {
    return abuNakhlaPrayerTimes;
  }

  try {
    // Abu Nakhla Health Centre area, Qatar.
    const latitude = ABU_NAKHLA_LAT;
    const longitude = ABU_NAKHLA_LNG;
    const url =
      `https://api.aladhan.com/v1/timings/${now.apiDate}` +
      `?latitude=${latitude}&longitude=${longitude}&method=8`;

    const response = await fetch(url);
    if (!response.ok) throw new Error('Prayer time request failed');

    const payload = await response.json();
    const timings = payload && payload.data && payload.data.timings;
    if (!timings) throw new Error('Prayer times unavailable');

    abuNakhlaPrayerTimes = {
      Fajr: formatPrayerTime12h(timings.Fajr),
      Dhuhr: formatPrayerTime12h(timings.Dhuhr),
      Asr: formatPrayerTime12h(timings.Asr),
      Maghrib: formatPrayerTime12h(timings.Maghrib),
      Isha: formatPrayerTime12h(timings.Isha)
    };
    abuNakhlaPrayerFetchDate = now.isoDate;
    return abuNakhlaPrayerTimes;
  } catch (error) {
    return null;
  }
}

function shuffleArray(items) {
  const arr = Array.isArray(items) ? items.slice() : [];
  for (let i = arr.length - 1; i > 0; i--) {
    let j;
    try {
      const random = new Uint32Array(1);
      crypto.getRandomValues(random);
      j = random[0] % (i + 1);
    } catch (e) {
      j = Math.floor(Math.random() * (i + 1));
    }
    const temp = arr[i];
    arr[i] = arr[j];
    arr[j] = temp;
  }
  return arr;
}

function getTickerDisplayDuration(text, minimumMs) {
  const chars = String(text || '').trim().length;

  // Allow comfortable reading time. Short items stay brisk; longer items remain
  // visible for proportionally longer instead of disappearing mid-read.
  const readingMs = chars * 85;

}

function initializeTickerSwipeHistory() {
  const ticker = document.getElementById('swapsNewsTicker');
  if (!ticker || ticker.dataset.swipeHistoryBound === '1') return;
  ticker.dataset.swipeHistoryBound = '1';

  ticker.addEventListener('touchstart', function(event) {
    const touch = event.touches && event.touches[0];
    if (!touch) return;
    rotasticTickerTouchStartX = touch.clientX;
    rotasticTickerTouchStartY = touch.clientY;
  }, {passive:true});

  ticker.addEventListener('touchend', function(event) {
    const touch = event.changedTouches && event.changedTouches[0];
    if (!touch) return;

    const dx = touch.clientX - rotasticTickerTouchStartX;
    const dy = touch.clientY - rotasticTickerTouchStartY;

    // Horizontal swipe only; vertical scrolling/pull-to-refresh remains untouched.
    if (Math.abs(dx) < 48 || Math.abs(dx) <= Math.abs(dy) * 1.35) return;

    if (dx > 0) navigateRotasticTickerHistory(-1); // swipe right = previous
    else navigateRotasticTickerHistory(1);        // swipe left = next/resume
  }, {passive:true});
}

function shuffleArray(items) {
  const arr = Array.isArray(items) ? items.slice() : [];
  for (let i = arr.length - 1; i > 0; i--) {
    let j;
    try {
      const random = new Uint32Array(1);
      crypto.getRandomValues(random);
      j = random[0] % (i + 1);
    } catch (e) {
      j = Math.floor(Math.random() * (i + 1));
    }
    const temp = arr[i];
    arr[i] = arr[j];
    arr[j] = temp;
  }
  return arr;
}

function refillRegularTickerDeck() {
  const messages = getRegularTickerMessages();

  const medicalRegional = [];
  const medicalInternational = [];
  const colleagueAdvice = [];

  messages.forEach(function(message) {
    const s = String(message || '').toLowerCase();
    const hasNumber = /\d/.test(message);

    if (!hasNumber) {
      colleagueAdvice.push(message);
    } else if (/qatar|doha|phcc|hamad|hmc|moph|gcc|gulf|saudi|uae|emirat|kuwait|bahrain|oman|mena|middle east|arab|jordan|egypt|lebanon|morocco|tunisia|iraq|turkey|turkish|pakistan|pakistani/.test(s)) {
      medicalRegional.push(message);
    } else {
      medicalInternational.push(message);
    }
  });

  const regional = shuffleArray(medicalRegional);
  const international = shuffleArray(medicalInternational);
  const advice = shuffleArray(colleagueAdvice);

  const fallbackAdvice = [
    '😊 Hope you have a smooth shift.',
    '💧 Hydrate when you get a quiet minute.',
    '✨ One patient at a time.',
    '☕ 2 quiet minutes can help reset.',
    '🤝 Keep the handover short and clear.',

    '📌 Highlight red flags clearly.',
    '📏 Correct cuff size matters.',
    '📖 Teach-back can reveal misunderstandings.',
    '🙌 Thank a colleague who helped.',
    '🧘 A calm tone helps in a busy queue.'
  ];

  const deck = [];
  let r = 0, i = 0, a = 0;
  let takeRegional = true;

  // Strict interleaving. Every medical/statistical item is followed by
  // a colleague-advice item before another medical fact can appear.
  while (r < regional.length || i < international.length) {
    let fact = '';

    // Qatar/GCC/Arab facts are strongly preferred. International facts
    // are occasional rather than clustered.
    if (r < regional.length && (takeRegional || i >= international.length)) {
      fact = regional[r++];
    } else if (i < international.length) {
      fact = international[i++];
    } else if (r < regional.length) {
      fact = regional[r++];
    }

    takeRegional = !takeRegional;

    if (fact) deck.push(fact);

    let adviceMessage = '';
    if (a < advice.length) {
      adviceMessage = advice[a++];
    } else {
      adviceMessage = fallbackAdvice[(a++) % fallbackAdvice.length];
    }
    deck.push(adviceMessage);
  }

  // If there are more advice items than facts, add them at the end.
  while (a < advice.length) deck.push(advice[a++]);

  regularTickerDeck = deck.length ? deck : shuffleArray(messages);
  regularTickerIndex = 0;
}

function getTrafficWindowForCurrentUser() {
  const shift = getActiveUserShiftWindowToday();
  if (!shift) return null;

  const now = getQatarClock().minutes;
  const start = shift.start;
  const end = shift.end;

  // Arrival window: 60 min before start until 20 min after start.
  if (now >= start - 60 && now <= start + 20) {
    return {
      phase:'arrival',
      shiftLabel:shift.label,
      minutesToEvent:start - now
    };
  }

  // End-of-shift window: 60 min before finish until 20 min after finish.
  if (now >= end - 60 && now <= end + 20) {
    return {
      phase:'departure',
      shiftLabel:shift.label,
      minutesToEvent:end - now
    };
  }

  return null;
}

function normalizeTrafficItem(item, phase) {
  if (!item) return null;

  if (typeof item === 'string') {
    return {text:item.trim(), phase:phase || '', source:''};
  }

  const text = String(item.text || item.summary || item.message || '').trim();
  const source = String(item.source || '').trim();
  const delayMinutes = Number(item.delayMinutes || item.delay || 0);

  if (!text) return null;

  return {
    text:text,
    phase:phase || String(item.phase || ''),
    source:source,
    delayMinutes:isNaN(delayMinutes) ? 0 : delayMinutes
  };
}

function shuffleArray(items) {
  const arr = Array.isArray(items) ? items.slice() : [];
  for (let i = arr.length - 1; i > 0; i--) {
    let j;
    try {
      const random = new Uint32Array(1);
      crypto.getRandomValues(random);
      j = random[0] % (i + 1);
    } catch (e) {
      j = Math.floor(Math.random() * (i + 1));
    }
    const temp = arr[i];
    arr[i] = arr[j];
    arr[j] = temp;
  }
  return arr;
}

function refillRegularTickerDeck() {
  const messages = getRegularTickerMessages();

  const medicalRegional = [];
  const medicalInternational = [];
  const colleagueAdvice = [];

  messages.forEach(function(message) {
    const s = String(message || '').toLowerCase();
    const hasNumber = /\d/.test(message);

    if (!hasNumber) {
      colleagueAdvice.push(message);
    } else if (/qatar|doha|phcc|hamad|hmc|moph|gcc|gulf|saudi|uae|emirat|kuwait|bahrain|oman|mena|middle east|arab|jordan|egypt|lebanon|morocco|tunisia|iraq|turkey|turkish|pakistan|pakistani/.test(s)) {
      medicalRegional.push(message);
    } else {
      medicalInternational.push(message);
    }
  });

  const regional = shuffleArray(medicalRegional);
  const international = shuffleArray(medicalInternational);
  const advice = shuffleArray(colleagueAdvice);

  const fallbackAdvice = [
    '😊 Hope you have a smooth shift.',
    '💧 Hydrate when you get a quiet minute.',
    '✨ One patient at a time.',
    '☕ 2 quiet minutes can help reset.',
    '🤝 Keep the handover short and clear.',
    '📝 Document the plan while it is fresh.',
    '🧠 Difficult case? Ask a colleague early.',
    '📞 Close the loop on important referrals.',
    '🩺 A quick medication check can prevent errors.',
    '🌿 Take a brief pause when the workload allows.',
    '📌 Highlight red flags clearly.',
    '📏 Correct cuff size matters.',
    '📖 Teach-back can reveal misunderstandings.',
    '🙌 Thank a colleague who helped.',
    '🧘 A calm tone helps in a busy queue.'
  ];

  const deck = [];
  let r = 0, i = 0, a = 0;
  let takeRegional = true;

  // Strict interleaving. Every medical/statistical item is followed by
  // a colleague-advice item before another medical fact can appear.
  while (r < regional.length || i < international.length) {
    let fact = '';

    // Qatar/GCC/Arab facts are strongly preferred. International facts
    // are occasional rather than clustered.
    if (r < regional.length && (takeRegional || i >= international.length)) {
      fact = regional[r++];
    } else if (i < international.length) {
      fact = international[i++];
    } else if (r < regional.length) {
      fact = regional[r++];
    }

    takeRegional = !takeRegional;

    if (fact) deck.push(fact);

    let adviceMessage = '';
    if (a < advice.length) {
      adviceMessage = advice[a++];
    } else {
      adviceMessage = fallbackAdvice[(a++) % fallbackAdvice.length];
    }
    deck.push(adviceMessage);
  }

  // If there are more advice items than facts, add them at the end.
  while (a < advice.length) deck.push(advice[a++]);

  regularTickerDeck = deck.length ? deck : shuffleArray(messages);
  regularTickerIndex = 0;
}

function getTrafficWindowForCurrentUser() {
  const shift = getActiveUserShiftWindowToday();
  if (!shift) return null;

  const now = getQatarClock().minutes;
  const start = shift.start;
  const end = shift.end;

  // Arrival window: 60 min before start until 20 min after start.
  if (now >= start - 60 && now <= start + 20) {
    return {
      phase:'arrival',
      shiftLabel:shift.label,
      minutesToEvent:start - now
    };
  }

  // End-of-shift window: 60 min before finish until 20 min after finish.
  if (now >= end - 60 && now <= end + 20) {
    return {
      phase:'departure',
      shiftLabel:shift.label,
      minutesToEvent:end - now
    };
  }

  return null;
}

function normalizeTrafficItem(item, phase) {
  if (!item) return null;

  if (typeof item === 'string') {
    return {text:item.trim(), phase:phase || '', source:''};
  }

  const text = String(item.text || item.summary || item.message || '').trim();
  const source = String(item.source || '').trim();
  const delayMinutes = Number(item.delayMinutes || item.delay || 0);

  if (!text) return null;

  return {
    text:text,
    phase:phase || String(item.phase || ''),
    source:source,
    delayMinutes:isNaN(delayMinutes) ? 0 : delayMinutes
  };
}
