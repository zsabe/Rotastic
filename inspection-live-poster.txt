riefing() {
  const greetingEl = document.getElementById('cozyGreetingText');
  const sentenceEl = document.getElementById('cozyDutySentence');
  const iconEl = document.getElementById('greetingIcon');
  const holidayLine = document.getElementById('holidayGreetingLine');

  const now = new Date();
  const qatarHour = parseInt(
    new Intl.DateTimeFormat('en-GB', {
      timeZone: 'Asia/Qatar',
      hour: '2-digit',
      hour12: false
    }).format(now),
    10
  );

  let iconClass = 'fa-solid fa-sun text-amber-400';

  if (qatarHour >= 12 && qatarHour < 17) {
    iconClass = 'fa-solid fa-cloud-sun text-amber-400';
  } else if (qatarHour >= 17) {
    iconClass = 'fa-solid fa-moon text-indigo-400';
  }

  const todayISO = getQatarTodayISO();
  const greetingContext = getRotasticGreetingContext(activeUserDocName);
  const holiday = greetingContext.holiday;

  if (greetingEl) {
    greetingEl.innerText = greetingContext.standardGreeting;
  }

  if (iconEl) {
    iconEl.className = 'inline-flex items-center justify-center w-6 text-base flex-shrink-0';
    iconEl.innerHTML = '<i class="' + iconClass + '"></i>';
  }

  if (holidayLine) {
    if (holiday) {
      holidayLine.innerText = holiday.greeting;
      holidayLine.className = 'text-xs mt-1.5 font-extrabold ' + holiday.className;
      holidayLine.classList.remove('hidden');
    } else {
      holidayLine.classList.add('hidden');
      holidayLine.innerText = '';
    }
  }

  updateAbuNakhlaWeather();

  if (!sentenceEl) return;

  if (!activeUserDocName) {
    sentenceEl.innerHTML =
      '👋 Select your physician profile to see your personal briefing for today.';
    return;
  }

  if (isDoctorOnLeaveToday(activeUserDocName, todayISO)) {
    sentenceEl.innerHTML =
      '<span class="text-emerald-300 font-bold">🏖️🌴 You are off today — enjoy your leave! 😎☀️</span> ' +
      '<span class="text-slate-400">Rest, recharge, and enjoy the day. 🍹✨</span>';
    return;
  }

  const today = getTodayScheduleDay();

  if (!today) {
    sentenceEl.innerHTML =
      '🩺 There is no published duty entry for you today. Enjoy the breathing space ☕🙂';
    return;
  }

  const briefing = chooseSingleDailyBriefing(today, activeUserDocName);

  if (!briefing) {
    sentenceEl.innerHTML =
      '✨ You have no assigned clinical duty on today’s published rota. Enjoy the lighter day! ☕🙂';
    return;
  }

  sentenceEl.innerHTML =
    '<span class="block mb-1"><strong class="text-white">' +
    briefing.title +
    '</strong> <span class="text-slate-300">' +
    briefing.details +
    '</span></span>';
}

function nextMedicalFact() {
  currentMedicalFactIndex = (currentMedicalFactIndex + 1) % WEIRD_MEDICAL_FACTS.length;
  renderMedicalFact();
}

function randomMedicalFact() {
  currentMedicalFactIndex = Math.floor(Math.random() * WEIRD_MEDICAL_FACTS.length);
  renderMedicalFact();
}

function renderMedicalFact() {
  const fact = WEIRD_MEDICAL_FACTS[currentMedicalFactIndex];
  const title = document.getElementById('medicalFactTitle');
  const text = document.getElementById('medicalFactText');

  if (title) title.innerText = fact.title;
  if (text) text.innerText = fact.fact;
}
