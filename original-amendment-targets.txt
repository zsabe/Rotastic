function getQatarRotaDayIndex() {
  // Prefer the exact Qatar date when the displayed rota contains today.
  // If this is a future/past prototype week, or today is the Qatar weekend
  // (Friday/Saturday), open Sunday rather than an arbitrary weekday.
  try {
    const parts = new Intl.DateTimeFormat('en-CA', {
      timeZone:'Asia/Qatar', year:'numeric', month:'2-digit', day:'2-digit'
    }).formatToParts(new Date());
    const value = {};
    parts.forEach(function(part){ if (part.type !== 'literal') value[part.type] = part.value; });
    const todayISO = [value.year, value.month, value.day].join('-');
    if (Array.isArray(currentSchedule) && currentSchedule.length) {
      const exact = currentSchedule.findIndex(function(day){ return day && day.isoDate === todayISO; });
      if (exact >= 0 && exact <= 4) return exact;
    }

    const weekday = new Intl.DateTimeFormat('en-GB', {
      timeZone:'Asia/Qatar', weekday:'short'
    }).format(new Date());
    const map = {Sun:0, Mon:1, Tue:2, Wed:3, Thu:4};
    return Object.prototype.hasOwnProperty.call(map, weekday) && (!currentSchedule || !currentSchedule.length)
      ? map[weekday]
      : 0;
  } catch (e) {
    if (Array.isArray(currentSchedule) && currentSchedule.length) {
      const local = new Date();
      const yyyy = String(local.getFullYear());
      const mm = String(local.getMonth()+1).padStart(2,'0');
      const dd = String(local.getDate()).padStart(2,'0');
      const exact = currentSchedule.findIndex(function(day){ return day && day.isoDate === yyyy+'-'+mm+'-'+dd; });
      if (exact >= 0 && exact <= 4) return exact;
      return 0;
    }
    const day = new Date().getDay();
    return day >= 0 && day <= 4 ? day : 0;
  }
}
