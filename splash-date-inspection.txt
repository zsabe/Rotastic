function getQatarLongDateText(isoDate) {
  try {
    const iso = isoDate || getQatarTodayISO();
    const parts = String(iso).split('-').map(Number);
    const d = new Date(Date.UTC(parts[0], parts[1] - 1, parts[2], 9, 0, 0));

    return new Intl.DateTimeFormat('en-GB', {
      timeZone:'Asia/Qatar',
      weekday:'long',
      day:'numeric',
      month:'long',
      year:'numeric'
    }).format(d);
  } catch (e) {
    return isoDate || getQatarTodayISO();
  }
}

function getQatarTodayISO() {
  try {
    const parts = new Intl.DateTimeFormat('en-CA', {
      timeZone: 'Asia/Qatar',
      year: 'numeric',
      month: '2-digit',
      day: '2-digit'
    }).formatToParts(new Date());

    const map = {};
    parts.forEach(function(p) {
      if (p.type !== 'literal') map[p.type] = p.value;
    });

    return map.year + '-' + map.month + '-' + map.day;
  } catch (e) {
    return new Date().toISOString().slice(0, 10);
  }
}

function rotasticPlannerMonthData(year, monthIndex) {
  const byDate=getAnnualPlannerEntriesByDate(year);
  const daysInMonth=new Date(year,monthIndex+1,0,12,0,0).getDate();
  const firstDow=new Date(year,monthIndex,1,12,0,0).getDay();
  const days=[];
  for(let d=1;d<=daysInMonth;d+=1){
    const iso=String(year)+'-'+String(monthIndex+1).padStart(2,'0')+'-'+String(d).padStart(2,'0');

  }

}
