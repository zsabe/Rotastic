async function rotasticRefreshGenerationLock() {
  try {
    if (rotasticBackendClient && rotasticBackendReady) {
      const result = await rotasticBackendClient.rpc('rotastic_rota_generation_lock_status');
      if (!result.error && result.data) rotasticGenerationLockCache = result.data;
    }
  } catch(e) { console.warn('ROTASTIC generation lock check:', e); }
  rotasticRenderGenerationLock();
  return rotasticGenerationLockInfo();
}

function rotasticGuardNewGeneration(showAlert) {
  const info = rotasticGenerationLockInfo();
  if (!info.locked) return true;
  if (showAlert !== false) {
    alert('This approved rota is locked. A new rota cannot be generated until ' + rotasticFormatGenerationUnlock(info.unlock_at) + '. Use the Smart Amendment box for changes to the current rota.');
  }
  return false;
}
