function getPersonalPosterTheme(leaveInfo) {
  if (!leaveInfo) return ROTASTIC_PERSONAL_POSTER_ART.off;
  const type = String(leaveInfo.type || '').toLowerCase();
  if (type === 'annual') return ROTASTIC_PERSONAL_POSTER_ART.annual;
  if (type === 'sick') return ROTASTIC_PERSONAL_POSTER_ART.sick;
  if (type === 'casual') return ROTASTIC_PERSONAL_POSTER_ART.casual;
  return Object.assign({}, ROTASTIC_PERSONAL_POSTER_ART.annual, {
    title:leaveInfo.label || 'Approved Leave',
    icon:leaveInfo.emoji || '🌿',
    message:'Thank you for your hard work. Rest, recharge and enjoy your time away. 🌿☀️'
  });
}
